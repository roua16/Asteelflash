  \item Added Employee to sidebar navigation (Dashboard + Materials Assignments).
  \item Added Employee role to AssignmentsInterface page (was Admin-only).

  \item All dashboards use the same unified style/layout.
\end{itemize}

\section{Roles \& Access Definition}

\subsection{Admin}
\begin{itemize}
  \item \textbf{Full access} to everything: all dashboards, materials, assignments, employees, suppliers, projects, offers, requests, delivery orders.
  \item \textbf{Administration section}: Materials, Assignments, Assignment Materials, Delivery Orders, Requests, Employees, Offers, Projects, Suppliers (CRUD admin pages).
  \item \textbf{Email}: receives all operation notifications (override: \texttt{SMTP\_ADMIN\_EMAIL}).
\end{itemize}

\subsection{PDR}
\begin{itemize}
  \item Full IT Inventory Dashboard (KPIs, quick actions, recent activity, charts).
  \item Materials View (PDR), Suppliers, Delivery Orders, Pending Deliveries, D.O History.
  \item Receives assignment return issue alerts.
\end{itemize}

\subsection{Purchasing}
\begin{itemize}
  \item Full IT Inventory Dashboard.
  \item Materials View, Purchase Requests, Suppliers, Delivery Orders, Pending Deliveries, D.O History.
  \item Archives: Archived Requests.
\end{itemize}

\subsection{IT}
\begin{itemize}
  \item Full IT Inventory Dashboard.
  \item Materials View, Materials Assignments, Infrastructure.
  \item Receives assignment return issue alerts.
\end{itemize}

\subsection{Infrastructure}
\begin{itemize}
  \item Full IT Inventory Dashboard.
  \item Materials Assignments, Infrastructure.
\end{itemize}

\subsection{Employee}
\begin{itemize}
  \item Employee Dashboard (limited view---own assignments, own requests, own recent activity, quick actions).
  \item Materials Assignments (view own assignments).
  \item Return Materials via Mission Interface.
  \item \textbf{No access to}: Administration, Purchase pages, Suppliers, Delivery Orders, Archives, Infrastructure.
\end{itemize}

\section{Testing \& How to Run}
\begin{itemize}
  \item \textbf{Unit tests}: \texttt{dotnet test} (UI tests skipped by default).
  \item \textbf{UI tests}: set \texttt{RUN\_UI\_TESTS=true}, optionally \texttt{PLAYWRIGHT\_CHANNEL=chrome}, then run \texttt{scripts/run-ui-tests.sh}.
  \item \textbf{Email testing}: project uses \texttt{smtp4dev} on local dev via docker compose.
\end{itemize}
